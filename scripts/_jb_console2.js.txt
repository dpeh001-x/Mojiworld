function jukeboxPlay(t) {
  if (!t || !t.file) return;
  // v0.29.73 — defense in depth: never play a track the player hasn't
  // discovered, even if a stale/queued click reaches here.
  if (typeof _isTrackHeard === 'function' && !_isTrackHeard(t.id)) {
    if (typeof showToast === 'function') showToast('🔒 That theme is still undiscovered.', 'rare');
    return;
  }
  // Stop any currently-playing jukebox preview
  jukeboxStop({silent:true});
  // Save the live BGM source ONCE so we can hand the auto-picker back
  // its track when the player closes / stops the jukebox.
  if (_jukeboxOriginalBgmSrc == null && typeof _bgmEl !== 'undefined' && _bgmEl) {
    _jukeboxOriginalBgmSrc = _bgmEl.src;
  }
  // Re-use the main _bgmEl element so volume + mute settings flow through
  // automatically. Set crossOrigin to keep things safe with raw.githack
  // hosting; let the browser handle file caching.
  if (typeof _bgmEl === 'undefined' || !_bgmEl) return;
  try { _bgmEl.pause(); } catch (_) {}
  _bgmEl.src = t.file;
  _bgmEl.loop = true;
  _bgmEl.currentTime = 0;
  const p = _bgmEl.play();
  if (p && p.catch) p.catch(() => {});
  _jukeboxPlaying = { trackId: t.id, name: t.name };
  _jbRefresh();   // v0.30.x jb-console — the pad lights, the record spins, the screen reads the track
}
function jukeboxStop(opts) {
  opts = opts || {};
  if (typeof _bgmEl !== 'undefined' && _bgmEl) {
    try { _bgmEl.pause(); } catch (_) {}
    // Restore the original BGM source so the auto-picker picks up where
    // it left off on the next map change.
    if (_jukeboxOriginalBgmSrc) {
      _bgmEl.src = _jukeboxOriginalBgmSrc;
      _jukeboxOriginalBgmSrc = null;
      // Resume the restored hub BGM on an EXPLICIT user Stop — pause()+src-restore
      // alone leaves it silent (only loadMap/_setBossBgm restart _bgmEl otherwise),
      // so town went silent on Stop. Skip on the silent map-change cleanup, where
      // loadMap sets the next map's BGM immediately after.
      const _muted = (typeof audio !== 'undefined' && audio && audio.muted) || (typeof game !== 'undefined' && game && game.muted);
      if (!opts.silent && !_muted) {
        try { _bgmEl.currentTime = 0; const _p = _bgmEl.play(); if (_p && _p.catch) _p.catch(() => {}); } catch (_) {}
      }
    }
  }
  _jukeboxPlaying = null;
  if (!opts.silent) _jbRefresh();   // v0.30.x jb-console
}
