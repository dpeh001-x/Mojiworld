  // Progress toward the next binding — top uncaptured species by kills.
  // v0.30.x mm-hunt — in the cute style of the cards above (per user: "do the same cute style for the hunt progress"):
  // a plush card, each species with its sprite in a pastel bubble, a candy progress bar with milestone dots, and its
  // count on a pill; a species you can bind now goes mint and says so. Same list, same order, same numbers.
  const prog = [];
  for (const k in game.bestiary) {
    if (k.charAt(0) === '_' || mm.roster[k] || !monsterTypes[k]) continue;
    const n = game.bestiary[k] | 0;
    if (n > 0) prog.push([k, n]);
  }
  prog.sort((a, b) => b[1] - a[1]);
  if (prog.length) {
    html += '<div class="mmh-card"><div class="mmh-h"><span class="mmh-hi">🐾</span><span class="mmh-ht">Hunt progress</span>' +
      '<em class="mmh-req">' + MOJIMON_KILLS_REQ.toLocaleString() + ' kills to bind</em></div><div class="mmh-list">';
    for (const [k, n] of prog.slice(0, 8)) {
      const t = monsterTypes[k];
      const pc = Math.min(1, n / MOJIMON_KILLS_REQ);
      const spr = (typeof _monsterDexSprite === 'function') ? _monsterDexSprite(k, t) : null;
      html += '<div class="mmh-row' + (pc >= 1 ? ' ready' : '') + '" data-mon="' + esc(k) + '">' +
        '<span class="mmh-bub">' + (spr ? '<img src="' + spr.src + '" alt="">' : '<i style="background:radial-gradient(circle at 35% 30%, ' + (t.shell || '#888') + ', ' + (t.color || '#666') + ');"></i>') + '</span>' +
        '<span class="mmh-name">' + esc(t.name || k) + '</span>' +
        '<span class="mmh-bar"><i style="width:' + (pc * 100).toFixed(1) + '%"></i></span>' +
        (pc >= 1 ? '<span class="mmh-count ready">⛓ Ready to bind!</span>'
                 : '<span class="mmh-count">' + n.toLocaleString() + ' / ' + MOJIMON_KILLS_REQ.toLocaleString() + '</span>') +
        '</div>';
    }
    html += '</div>';
    if (prog.length && prog[0][1] >= MOJIMON_KILLS_REQ) {
      html += '<div class="mmh-go">✨ Eligible species found — get close to a wild one to bind it!</div>';
    }
    html += '</div>';
  }
