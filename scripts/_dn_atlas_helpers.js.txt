// v0.30.x dn-atlas — THE POP, FROM A GLYPH ATLAS (see the note at the live passes in drawDamageNumbers).
// A popping number rasterised 3-6 passes of thick stroked text at a size that changes every frame, and numbers
// of several base sizes pop at once, so the glyph cache never had the size warm: measured, the pop was ~80% of
// the canvas flush of a fight with hits landing. The twelve glyphs a damage figure is made of are rendered
// ONCE per (base size, pop size, style, colour), at that exact font size with a literal 5 px outline - so the
// outline is the same constant it is in every other phase - and a popping number is a few blits.
// One ROW PER PASS of the live path (shadow, halo, outline, rim, fill + highlight): drawn row by row across the
// whole number, a digit's halo can never tint its neighbour's outline, exactly as when each pass was one call.
let _LX_DN_ATLAS_ON = true;
const _LX_DN_ATLAS_CHARS = '0123456789,★';
const _LX_DN_ATLAS_OK = /^[0-9,]+★?$/;
const _LX_DN_ATLAS = new Map();            // key -> atlas; insertion order is the LRU order
const _LX_DN_ATLAS_MAX_PX = 12e6;          // ~48 MB of backing store, oldest out first
let _lxDnAtlasPx = 0;
let _lxDnAtlasBudget = 0;                  // builds still allowed this frame (drawDamageNumbers resets it)
// The pop's size, snapped to a 4% ladder hung on the base size (so scale 1 is exactly the base) and then to the
// half-pixel grid: the four frames around the apex share one atlas, and a pop needs six or seven, not ten.
const _LX_DN_ATLAS_LN = Math.log(1.04);
function _lxDnAtlasPx4(b0, scale) {
  const k = Math.round(Math.log(Math.max(0.05, scale)) / _LX_DN_ATLAS_LN);
  return Math.max(4, Math.round(b0 * Math.pow(1.04, k) * 2) / 2);
}
function _lxDnAtlasGet(b0, px, baseSize, d, col, tk, lowFx, stress, dpr) {
  const key = b0 + '|' + px + '|' + (d.crit ? 1 : 0) + (d.big ? 1 : 0) + (tk ? 1 : 0) + '|' + col + '|' + (lowFx ? 1 : 0) + (stress ? 1 : 0) + '|' + dpr;
  let at = _LX_DN_ATLAS.get(key);
  if (at !== undefined) {
    if (at) { _LX_DN_ATLAS.delete(key); _LX_DN_ATLAS.set(key, at); }   // most recently used goes last
    return at;
  }
  if (_lxDnAtlasBudget <= 0) return null;   // not this frame: the caller draws live, as it always did
  _lxDnAtlasBudget--;
  at = _lxDnAtlasBuild(b0, px, baseSize, d, col, tk, lowFx, stress, dpr);
  _LX_DN_ATLAS.set(key, at);                // a failed build is remembered as null, so it is not retried every frame
  if (at) {
    _lxDnAtlasPx += at.px;
    for (const k of _LX_DN_ATLAS.keys()) {
      if (_lxDnAtlasPx <= _LX_DN_ATLAS_MAX_PX || k === key) break;
      const old = _LX_DN_ATLAS.get(k); _LX_DN_ATLAS.delete(k);
      if (old) { _lxDnAtlasPx -= old.px; old.cv.width = old.cv.height = 0; }
    }
  }
  return at;
}
function _lxDnAtlasBuild(b0, px, baseSize, d, col, tk, lowFx, stress, dpr) {
  try {
    const S = px / Math.max(1, b0);                                  // the pop scale this size stands for
    const crit = !!d.crit, grad = (crit || d.big || tk) && !stress;   // the same gates the live passes use
    const halo = grad, rim = (crit || tk) && !lowFx && !stress, hi = (crit || d.big || tk) && !lowFx && !stress;
    const cv = document.createElement('canvas'); const c = cv.getContext('2d');
    const font = '900 ' + px + 'px Impact, "Arial Black", "Trebuchet MS", sans-serif';
    c.font = font;
    const mAll = c.measureText(_LX_DN_ATLAS_CHARS);
    const asc = Math.ceil(mAll.actualBoundingBoxAscent || px), desc = Math.ceil(mAll.actualBoundingBoxDescent || px * 0.3);
    const pad = Math.ceil(Math.max(halo ? (crit ? 9 : 7) * S / 2 : 0, 2.5) + 3 * S + 2);   // the widest stroke's reach + the shadow offset
    const base = asc + pad, cellHCss = base + desc + pad;
    const cellH = Math.ceil(cellHCss * dpr), rowH = cellH + 2;
    const cells = new Map(); let x = 0;
    for (const ch of _LX_DN_ATLAS_CHARS) {
      const adv = c.measureText(ch).width; const sw = Math.ceil((Math.ceil(adv) + pad * 2) * dpr);
      cells.set(ch, { sx: x, sw, adv }); x += sw + 2;
    }
    const passes = ['shadow']; if (halo) passes.push('halo'); passes.push('outline'); if (rim) passes.push('rim'); passes.push('fill');
    cv.width = x; cv.height = rowH * passes.length;
    let fillStyle;
    if (grad) {
      fillStyle = c.createLinearGradient(0, -px * 0.9, 0, px * 0.25);
      if (crit) { fillStyle.addColorStop(0, '#fff4b8'); fillStyle.addColorStop(0.45, '#ffd84a'); fillStyle.addColorStop(1, '#ff8a1f'); }
      else { fillStyle.addColorStop(0, _mixHex(col, '#ffffff', 0.55)); fillStyle.addColorStop(1, col); }
    } else if (crit) fillStyle = '#ffd84a';
    else if (!stress && !lowFx) fillStyle = _lxPlainGrad(c, col, Math.max(1, Math.round(px)));
    else fillStyle = col;
    const bsS = baseSize * S;   // the highlight band is cut in the live path's (float) base size
    for (let r = 0; r < passes.length; r++) {
      const pass = passes[r];
      for (const [ch, cell] of cells) {
        c.save();
        c.setTransform(1, 0, 0, 1, 0, 0);
        c.beginPath(); c.rect(cell.sx, r * rowH, cell.sw, cellH); c.clip();   // a glyph never paints into its neighbour's cell
        c.setTransform(dpr, 0, 0, dpr, cell.sx + pad * dpr, r * rowH + base * dpr);
        c.font = font; c.textAlign = 'left'; c.textBaseline = 'alphabetic'; c.lineJoin = 'round'; c.lineCap = 'round';
        if (pass === 'shadow') { c.globalAlpha = 0.55; c.fillStyle = '#000'; c.fillText(ch, 2 * S, 3 * S); }
        else if (pass === 'halo') { c.lineWidth = (crit ? 9 : 7) * S; c.strokeStyle = col; c.globalAlpha = 0.35; c.strokeText(ch, 0, 0); }
        else if (pass === 'outline') { c.lineWidth = 5; c.strokeStyle = '#000'; c.strokeText(ch, 0, 0); }   // THE readability anchor: a literal 5, at every pop size
        else if (pass === 'rim') { c.lineWidth = 2 * S; c.strokeStyle = crit ? '#fff7b0' : _mixHex(col, '#ffffff', 0.72); c.strokeText(ch, 0, 0); }
        else {
          c.fillStyle = fillStyle; c.fillText(ch, 0, 0);
          if (hi) {
            c.globalAlpha = crit ? 0.7 : 0.4;
            c.beginPath(); c.rect(-bsS * 2.2, -bsS * 0.95, bsS * 4.4, bsS * 0.38); c.clip();
            c.fillStyle = '#ffffff'; c.fillText(ch, 0, 0);
          }
        }
        c.restore();
      }
    }
    return { cv, cells, pad, base, cellH, rowH, rows: passes.length, dpr, px: cv.width * cv.height };
  } catch (e) { return null; }
}
// Blit one number from its atlas. The context arrives translated to the number's anchor, rotated if it is
// wobbling, and scaled by the pop; the glyphs are already the popped size, so the scale is cancelled. With no
// rotation the blits land on whole device pixels at exactly their own size - no resample, the same guarantee
// the settled bitmap carries (v0.30.466). Returns false (draw live) if a glyph is missing or the matrix is unreadable.
function _lxDnAtlasBlit(at, txt, scale, rotated) {
  const n = txt.length; let wTot = 0;
  for (let i = 0; i < n; i++) { const c0 = at.cells.get(txt[i]); if (!c0) return false; wTot += c0.adv; }
  const dpr = at.dpr;
  if (!rotated) {
    const tf = (typeof ctx.getTransform === 'function') ? ctx.getTransform() : null;
    if (!tf) return false;
    ctx.save();
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    const oy = Math.round(tf.f - at.base * dpr);
    for (let r = 0; r < at.rows; r++) {
      let pen = -wTot / 2;
      for (let i = 0; i < n; i++) {
        const c0 = at.cells.get(txt[i]);
        ctx.drawImage(at.cv, c0.sx, r * at.rowH, c0.sw, at.cellH, Math.round(tf.e + (pen - at.pad) * dpr), oy, c0.sw, at.cellH);
        pen += c0.adv;
      }
    }
    ctx.restore();
    return true;
  }
  ctx.save();
  if (scale !== 1) ctx.scale(1 / scale, 1 / scale);
  for (let r = 0; r < at.rows; r++) {
    let pen = -wTot / 2;
    for (let i = 0; i < n; i++) {
      const c0 = at.cells.get(txt[i]);
      ctx.drawImage(at.cv, c0.sx, r * at.rowH, c0.sw, at.cellH, pen - at.pad, -at.base, c0.sw / dpr, at.cellH / dpr);
      pen += c0.adv;
    }
  }
  ctx.restore();
  return true;
}
