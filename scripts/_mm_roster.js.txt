  // v0.30.x mm-roster — the roster cards in the cute style of the cards above (per user: "do the same cute style for
  // the roster cards"): a plush card per MojiMon, its sprite in a pastel bubble (bobbing while it is fielded), its
  // stats as pills, ★ H and Summon as candy buttons, the bond as a pill, and the point rows as candy steppers with a
  // little meter up to the per-stat cap. Same handlers, same numbers. Stylesheet: _mmRosterCss().
  if (!rosterKeys.length) {
    html += '<div class="mmr-empty"><span class="mmr-empty-ico"><i class="mmr-snail" aria-hidden="true" style="display:block;width:70px;height:49px;background:url(Sprites/monsters/snail.webp) -5.98px -29.66px/78.68px 78.68px no-repeat;filter:drop-shadow(1px 0px 0 #140c18) drop-shadow(-1px 0px 0 #140c18) drop-shadow(0px 1px 0 #140c18) drop-shadow(0px -1px 0 #140c18) drop-shadow(1.6px 0px 0 #fbeaff) drop-shadow(-1.6px 0px 0 #fbeaff) drop-shadow(0px 1.6px 0 #fbeaff) drop-shadow(0px -1.6px 0 #fbeaff) drop-shadow(0 3px 3px rgba(0,0,0,.45))"></i></span><div><b class="mmr-et">No MojiMon bound yet</b><span class="mmr-es">your first buddy is out there waiting</span></div></div>';
  } else {
    html += '<div class="mmr-grid">';
    for (const k of rosterKeys) {
      const t = monsterTypes[k] || {};
      const rec = mm.roster[k];
      const st = _mojimonStatsFor(k);
      const spr = (typeof _monsterDexSprite === 'function') ? _monsterDexSprite(k, t) : null;
      const free = pts - _mojimonSpent(rec);
      const isOut = mm.out && mm.out.type === k;
      const isH = mm.assigned === k;
      const row = (stat, label, valTxt) => {
        const v = rec.upg[stat] | 0, full = (free <= 0 || v >= MOJIMON_UPG_PT_CAP);
        return '<div class="mmr-row ' + stat + '">' +
          '<span class="mmr-lab">' + label + '</span>' +
          '<button type="button" class="mmr-step minus" onclick="_mojimonUi.upg(\'' + k + '\',\'' + stat + '\',-1)"' + (v <= 0 ? ' disabled' : '') + ' aria-label="' + label + ' down">−</button>' +
          '<b class="mmr-val">' + v + '</b>' +
          '<button type="button" class="mmr-step plus" onclick="_mojimonUi.upg(\'' + k + '\',\'' + stat + '\',1)"' + (full ? ' disabled' : '') + ' aria-label="' + label + ' up">+</button>' +
          '<span class="mmr-meter" title="' + v + ' of ' + MOJIMON_UPG_PT_CAP + '"><i style="width:' + Math.round(Math.min(1, v / MOJIMON_UPG_PT_CAP) * 100) + '%"></i></span>' +
          '<span class="mmr-bonus">' + valTxt + '</span></div>';
      };
      html += '<div class="mmr-card' + (isOut ? ' out' : '') + (isH ? ' h' : '') + '" data-mon="' + esc(k) + '">' +
        '<div class="mmr-top">' +
        '<div class="mmr-bub">' + (spr ? '<img src="' + spr.src + '" alt="">' : '<i style="background:radial-gradient(circle at 35% 30%, ' + (t.shell || '#888') + ', ' + (t.color || '#666') + ');"></i>') + '</div>' +
        '<div class="mmr-id"><div class="mmr-name">' + esc(t.name || k) + (isOut ? '<span class="mmr-out">● FIELDED</span>' : '') + '</div>' +
        '<div class="mmr-stats"><span class="mmr-sp hp" title="Max HP">💖 ' + st.maxHp.toLocaleString() + '</span><span class="mmr-sp atk" title="Attack">⚔ ' + st.atk.toLocaleString() + '</span><span class="mmr-sp def" title="Damage reduction">🛡 ' + Math.round(st.defRed * 100) + '%</span></div>' +
        (isOut && mm.out && mm.out.hpFrac != null ? '<div class="mmr-hp"><i style="width:' + Math.round(Math.max(0, Math.min(1, mm.out.hpFrac)) * 100) + '%"></i></div>' : '') +
        '</div>' +
        '<button type="button" class="mmr-hkey' + (isH ? ' on' : '') + '" title="Assign to the H quick-summon hotkey" onclick="_mojimonUi.assign(\'' + k + '\')">' + (isH ? '★ H' : '☆ H') + '</button>' +
        '</div>' +
        (isOut ? '' : '<button type="button" class="mmr-summon" onclick="_mojimonUi.summon(\'' + k + '\')"' + (cdLeft > 0 ? ' disabled' : '') + '>' + (cdLeft > 0 ? '💤 Summon ' + esc(t.name || k) + ' after the rest' : '✨ Summon ' + esc(t.name || k)) + '</button>') +
        // v0.30.713 — what this species lends you while it holds the H slot. Shown on every row, so
        // the roster reads as a menu of bonds rather than a list of pets; lit on the assigned one,
        // because that is the only one paying out.
        ((function () {
          const _al = _mojimonAuraLabel(k);
          return '<div class="mmr-bond' + (isH ? ' on' : '') + '" title="' + (_al.level ? 'Species level ' + _al.level : 'Species level unknown — this bond sits at the floor')
            + '. A bond runs from +0.1% at Lv 1 to +2.5% at Lv 100, and only the assigned MojiMon pays it.">'
            + '<span class="mmr-bk">🔗 ' + (isH ? 'BOND' : 'bond') + '</span>'
            + '<span class="mmr-bt">' + _al.text + '</span>'
            + '<span class="mmr-bl">' + (_al.level ? 'Lv ' + _al.level : 'Lv ?') + '</span>'
            + '</div>';
        })()) +
        '<div class="mmr-alloc">' +
        '<div class="mmr-ah"><span class="mmr-at">✨ Allocate points</span><em class="mmr-chip' + (free > 0 ? ' free' : '') + '">' + free + ' free</em><em class="mmr-chip">max ' + MOJIMON_UPG_PT_CAP + ' per stat</em></div>' +
        row('hp', 'HP', '+' + Math.round(_mmUpgPts(rec.upg.hp) * MOJIMON_UPG.hp * 100) + '%') +
        row('atk', 'ATK', '+' + Math.round(_mmUpgPts(rec.upg.atk) * MOJIMON_UPG.atk * 100) + '%') +
        row('def', 'DEF', Math.round(Math.min(MOJIMON_DEF_CAP, _mmUpgPts(rec.upg.def) * MOJIMON_UPG.def) * 100) + '% DR') +
        '</div></div>';
    }
    html += '</div>';
  }
