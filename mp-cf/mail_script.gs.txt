// Mojiworld feedback mailer. Runs as the Google account that deploys it (admin@moji-studios.com)
// and mails each player report to a moji-studios.com address.
// Deploy: Deploy -> New deployment -> type "Web app" -> Execute as: Me -> Who has access: Anyone -> Deploy.
// The game server calls  <web app address>?k=<KEY>  - requests without the right key are refused.
var KEY = '__KEY__';
function doPost(e) {
  try {
    if (!KEY || !e.parameter || e.parameter.k !== KEY) return out({ ok: false, error: 'bad key' });
    var d = JSON.parse(e.postData.contents);
    var to = String(d.to || '');
    if (!/^[^\s@]+@moji-studios\.com$/.test(to)) return out({ ok: false, error: 'recipient not allowed' });   // even with the key it can only mail the studio
    var o = { to: to, subject: String(d.subject || 'Mojiworld report').slice(0, 200), body: String(d.body || '').slice(0, 20000), name: 'Mojiworld Feedback' };
    if (/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(String(d.replyTo || ''))) o.replyTo = String(d.replyTo);
    if (d.shot) o.attachments = [Utilities.newBlob(Utilities.base64Decode(String(d.shot)), 'image/jpeg', 'report-' + String(d.id || '').slice(-6) + '.jpg')];
    MailApp.sendEmail(o);
    return out({ ok: true });
  } catch (err) { return out({ ok: false, error: String(err).slice(0, 200) }); }
}
function out(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
